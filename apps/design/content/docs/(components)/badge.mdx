## Installation

```bash
bunx shadcn@latest add badge
```

## Usage

```tsx
import { Badge } from "@/components/ui/badge";
```

```tsx
<Badge variant="success">Badge</Badge>
```

## Composition

Import the primitive that matches the badge anatomy you need.

```tsx
import { SparklesIcon } from "lucide-react";

import { Avatar, AvatarFallback } from "@/components/ui/avatar";
import { Badge, IconBadge, StatusBadge, UserBadge } from "@/components/ui/badge";
```

```tsx
<Badge variant="information">Badge</Badge>
<StatusBadge status="success">Live</StatusBadge>
<IconBadge variant="feature" aria-label="Feature">
  <SparklesIcon />
</IconBadge>
<UserBadge>
  <Avatar size="2xs">
    <AvatarFallback className="bg-[#18181b] text-white">M</AvatarFallback>
  </Avatar>
  Manik
</UserBadge>
```
