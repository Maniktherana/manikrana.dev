## Installation

```bash
bunx shadcn@latest add alert
```

## Usage

```tsx
import {
  Alert,
  AlertDescription,
  AlertTitle,
} from "@/components/ui/alert";
```

```tsx
<Alert status="information">
  <AlertTitle>Heads up</AlertTitle>
  <AlertDescription>New inventory data is available.</AlertDescription>
</Alert>
```

## API Reference

### Alert

| Prop | Type | Default |
| --- | --- | --- |
| `status` | `"neutral" \| "information" \| "success" \| "warning" \| "error"` | `"neutral"` |
| `variant` | `"default" \| "destructive"` | - |
| `children` | `React.ReactNode` | - |
| `className` | `string` | - |

### AlertTitle

| Prop | Type | Default |
| --- | --- | --- |
| `children` | `React.ReactNode` | - |
| `className` | `string` | - |

### AlertDescription

| Prop | Type | Default |
| --- | --- | --- |
| `children` | `React.ReactNode` | - |
| `className` | `string` | - |

### AlertAction

| Prop | Type | Default |
| --- | --- | --- |
| `children` | `React.ReactNode` | - |
| `className` | `string` | - |
