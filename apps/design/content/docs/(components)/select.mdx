## Installation

```bash
bunx shadcn@latest add select
```

## Usage

```tsx
import {
  Select,
  SelectContent,
  SelectGroup,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@/components/ui/select";
```

```tsx
<Select defaultValue="us">
  <SelectTrigger>
    <SelectValue placeholder="Choose region" />
  </SelectTrigger>
  <SelectContent>
    <SelectGroup>
      <SelectItem value="us">United States</SelectItem>
      <SelectItem value="in">India</SelectItem>
    </SelectGroup>
  </SelectContent>
</Select>
```
