## Installation

```bash
bunx shadcn@latest add hovercard
```

## Usage

```tsx
import { HoverCard, HoverCardContent, HoverCardTrigger } from "@/components/ui/hover-card";
```

```tsx
<HoverCard>
  <HoverCardTrigger render={<button type="button" />}>Open preview</HoverCardTrigger>
  <HoverCardContent>Preview content</HoverCardContent>
</HoverCard>
```
