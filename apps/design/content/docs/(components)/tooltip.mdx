## Installation

```bash
bunx shadcn@latest add tooltip
```

## Usage

```tsx
import {
  Tooltip,
  TooltipContent,
  TooltipTrigger,
} from "@/components/ui/tooltip";
```

```tsx
<Tooltip>
  <TooltipTrigger render={<button type="button" />}>Hover me</TooltipTrigger>
  <TooltipContent>Compact contextual help</TooltipContent>
</Tooltip>
```

## API Reference

### Tooltip

| Prop | Type | Default |
| --- | --- | --- |
| `open` | `boolean` | - |
| `defaultOpen` | `boolean` | - |
| `onOpenChange` | `(open: boolean) => void` | - |

### TooltipTrigger

| Prop | Type | Default |
| --- | --- | --- |
| `render` | `ReactElement \| (props, state) => ReactElement` | - |
| `className` | `string` | - |

### TooltipContent

| Prop | Type | Default |
| --- | --- | --- |
| `side` | `"top" \| "right" \| "bottom" \| "left" \| "inline-start" \| "inline-end"` | `"top"` |
| `sideOffset` | `number` | `4` |
| `align` | `"start" \| "center" \| "end"` | `"center"` |
| `alignOffset` | `number` | `0` |
| `className` | `string` | - |
