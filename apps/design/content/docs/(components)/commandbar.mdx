## Installation

```bash
bunx shadcn@latest add command
```

## Usage

```tsx
import { CommandBar, CommandBarAction } from "@/components/ui/command";
```

```tsx
<CommandBar selectedLabel="3 selected">
  <CommandBarAction shortcuts={["Cmd", "A"]}>Archive</CommandBarAction>
  <CommandBarAction shortcuts={["Cmd", "E"]}>Export</CommandBarAction>
</CommandBar>
```

## API Reference

### CommandBar

| Prop | Type | Default |
| --- | --- | --- |
| `selectedLabel` | `string` | `"1 selected"` |
| `className` | `string` | - |
| `children` | `ReactNode` | - |

### CommandBarAction

| Prop | Type | Default |
| --- | --- | --- |
| `shortcuts` | `string[]` | `["⌘", "⌘"]` |
| `edge` | `boolean` | `false` |
| `className` | `string` | - |
| `children` | `ReactNode` | - |
