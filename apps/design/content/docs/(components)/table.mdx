## Installation

```bash
bunx shadcn@latest add table
```

## Usage

```tsx
import {
  Table,
  TableBody,
  TableCaption,
  TableCell,
  TableFooter,
  TableHead,
  TableHeader,
  TableRow,
} from "@/components/ui/table";
```

```tsx
<Table>
  <TableHeader>
    <TableRow>
      <TableHead>Component</TableHead>
      <TableHead>Status</TableHead>
    </TableRow>
  </TableHeader>
  <TableBody>
    <TableRow>
      <TableCell>Button</TableCell>
      <TableCell>Ready</TableCell>
    </TableRow>
  </TableBody>
</Table>
```
