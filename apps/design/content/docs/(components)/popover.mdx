## Installation

```bash
bunx shadcn@latest add popover
```

## Usage

```tsx
import {
  Popover,
  PopoverContent,
  PopoverDescription,
  PopoverFooter,
  PopoverFooterActions,
  PopoverFooterButton,
  PopoverHeader,
  PopoverSeparator,
  PopoverStep,
  PopoverTitle,
  PopoverTrigger,
} from "@/components/ui/popover";
```

```tsx
<Popover>
  <PopoverTrigger>Open popover</PopoverTrigger>
  <PopoverContent showArrow showCloseButton>
    <PopoverHeader>
      <PopoverTitle>Insert Popover</PopoverTitle>
      <PopoverDescription>
        Insert popover description here. It would look much better as three lines of text.
      </PopoverDescription>
    </PopoverHeader>
    <PopoverSeparator />
    <PopoverFooter>
      <PopoverStep>Step 1 of 5</PopoverStep>
      <PopoverFooterActions>
        <PopoverFooterButton>Back</PopoverFooterButton>
        <PopoverFooterButton tone="primary" stretch>
          Next
        </PopoverFooterButton>
      </PopoverFooterActions>
    </PopoverFooter>
  </PopoverContent>
</Popover>
```

## API Reference

### Popover

`Popover` is the Base UI popover root.

| Prop | Type | Default |
| --- | --- | --- |
| `defaultOpen` | `boolean` | - |
| `open` | `boolean` | - |
| `onOpenChange` | `(open: boolean) => void` | - |

### PopoverContent

| Prop | Type | Default |
| --- | --- | --- |
| `side` | `"top" \| "bottom" \| "left" \| "right" \| "inline-start" \| "inline-end"` | `"bottom"` |
| `sideOffset` | `number` | `4` |
| `align` | `"start" \| "center" \| "end"` | `"center"` |
| `alignOffset` | `number` | `0` |
| `showArrow` | `boolean` | `false` |
| `showCloseButton` | `boolean` | `false` |
| `className` | `string` | - |

### PopoverFooterButton

| Prop | Type | Default |
| --- | --- | --- |
| `tone` | `"muted" \| "primary"` | `"muted"` |
| `stretch` | `boolean` | `false` |
| `size` | `"default" \| "xs" \| "sm" \| "lg" \| "icon" \| "icon-xs" \| "icon-sm" \| "icon-lg"` | `"sm"` |
| `className` | `string` | - |
