## Installation

```bash
bunx shadcn@latest add avatar
```

## Usage

```tsx
import { Avatar, AvatarFallback, AvatarImage } from "@/components/ui/avatar";
```

```tsx
<Avatar>
  <AvatarImage src="/avatar.png" alt="Manik Rana" />
  <AvatarFallback>MR</AvatarFallback>
</Avatar>
```
