## Installation

```bash
bunx shadcn@latest add kbd
```

## Usage

```tsx
import { Kbd, KbdGroup } from "@/components/ui/kbd";

export function KbdDemo() {
  return (
    <KbdGroup>
      <Kbd>Cmd</Kbd>
      <Kbd>K</Kbd>
    </KbdGroup>
  );
}
```
